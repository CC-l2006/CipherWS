# CipherWS 前端更新（服务器：/opt/cipherws）

# ---- 推荐：用仓库里的脚本，一条命令搞定 ----
cd /opt/cipherws
./deploy/update-frontend.sh

# 脚本会依次做：环境自检 → git pull → 补 .env.production → 按需装依赖
#              → 构建两个站点 → 重启主站服务 → 同步 nginx（先校验后加载）→ 自检

# 常用变体：
#   ./deploy/update-frontend.sh --site main    只更新主站
#   ./deploy/update-frontend.sh --site train   只更新云间列车（无需重启服务）
#   ./deploy/update-frontend.sh --skip-pull    改了 .env.production 后重新构建
#   ./deploy/update-frontend.sh --ci           依赖有变动时用 npm ci
#   ./deploy/update-frontend.sh --no-restart   只构建，不动服务
#   ./deploy/update-frontend.sh --help


# ---- 等价的手工命令（脚本出问题时用）----

# 只更新其中一个站点：
#   npm run build:main   → 只重建主站（需重启 cipherws-web-main）
